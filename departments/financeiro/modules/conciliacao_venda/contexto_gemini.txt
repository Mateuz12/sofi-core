from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.action_chains import ActionChains
import time

# ==========================================
# 1. O CÉREBRO: LENDO A PLANILHA (PANDAS)
# ==========================================
# Coloque o nome exato da sua planilha de vendas aqui
df_vendas = pd.read_excel('vendas.xlsx', decimal=',')

pedido_alvo = 497833698
doc_alvo = "434962754" # O 'Doc' que aparece na caixa laranja da sua venda

# Lembre-se de colocar o nome da coluna da NSU aqui:
coluna_nsu = 'NOME_DA_COLUNA_AQUI' 
lista_nsus = df_vendas[df_vendas['NumeroPedido'] == pedido_alvo][coluna_nsu].dropna().astype(str).tolist()

print(f"Alvo Laranja (Doc): {doc_alvo}")
print(f"Alvos Azuis (NSUs): {lista_nsus}")
print("-" * 30)

# ==========================================
# 2. OS OLHOS: ACESSANDO O SISTEMA (SELENIUM)
# ==========================================
navegador = webdriver.Chrome()

# Substitua pela URL exata do Finanças GB
navegador.get("https://COLOQUE_A_URL_DO_FINANCAS_AQUI.com.br")

# O robô entra em modo de espera para você fazer o login e chegar na tela certa
input("HUMANO: Faça o login, vá até a tela de Conciliação e aperte ENTER aqui no terminal para eu assumir...")

# ==========================================
# 3. AS MÃOS: ARRASTAR E SOLTAR (ACTIONCHAINS)
# ==========================================
garra = ActionChains(navegador)

try:
    # O robô procura na tela qualquer bloco que contenha o texto do Doc (Caixa Laranja)
    print("Procurando a caixa laranja...")
    caixa_laranja = navegador.find_element(By.XPATH, f"//*[contains(text(), '{doc_alvo}')]")

        print(f"Procurando a caixa azul da NSU {nsu}...")
        caixa_azul = navegador.find_element(By.XPATH, f"//*[contains(text(), '{nsu}')]")
        
        print(f"Arrastando NSU {nsu} para o Doc {doc_alvo}...")
        # Comando mágico de Drag and Drop: Origem, Destino, Executar.
        garra.drag_and_drop(caixa_azul, caixa_laranja).perform()
        
        # Pausa de 1 segundo para o sistema da indústria não achar que é um ataque de servidor
        time.sleep(1) 
        
    print("\nSUCESSO: Conciliação das parcelas executada!")
    
except Exception as e:
    print(f"\nERRO: O robô não conseguiu encontrar as caixas. Verifique se elas estão visíveis na tela.")
    print(f"Detalhe do erro técnico para investigarmos: {e}")

input("Aperte ENTER para fechar o robô...")

A Solução de Engenharia: "Memória de Cascata" (Forward Fill)
Para resolver isso, nós vamos usar um dos comandos mais poderosos do Pandas: o ffill() (Forward Fill).

Nós vamos ensinar o robô a fazer exatamente o que o seu olho faz: ele vai ler a linha "Loja 20845", vai guardar esse número na memória e "empurrar" ele para a linha de baixo. Depois, ele filtra apenas as linhas que estão escritas "001 - DINHEIRO".

# 1. O robô procura o texto que começa com "Loja:" e extrai só o número dela.

Se nós não temos uma informação 100% única (NSU), nós pegamos duas ou três informações "fracas" e juntamos para criar uma "impressão digital" forte.

# Transforma Valor em Número de verdade para podermos somar
def converter_para_float(valor):
    if pd.isna(valor): 
        return 0.0
    
    # TRAVA DE SEGURANÇA: Se o Pandas já leu como número, não mexa na pontuação!
    if isinstance(valor, (int, float)):
        return float(valor)
    
    # Se o Pandas leu como Texto (string), nós fazemos a limpeza brasileira
    v_str = str(valor).strip()
    v_str = v_str.replace('.', '').replace(',', '.')
    
    try: 
        return float(v_str)
    except ValueError: 
        return 0.0

def converter_para_float(valor):
    if pd.isna(valor):

    # Se o Pandas já leu como número, não mexe
    if isinstance(valor, (int, float)):
        return float(valor)
    
    # Se leu como texto, faz a limpeza
    v_str = str(valor).strip().replace('.', '').replace(',', '.')
    try: 
        return float(v_str)
    except ValueError: 
        return 0.0

A Matemática: Primeiro, ele soma os valores de todas as caixas azuis daquele pedido e guarda na memória o Valor Total (Ex: R$ 437,49).

O Diagnóstico: O "Míope" vs O "Sniper"
No seu MVP, você usava este comando:

Isso dizia para o robô: "Ache o texto exato da NSU e clique EXATAMENTE em cima da palavra". Era um tiro de sniper.

Mateus, agora o cenário está cristalino. Se ele não selecionou a caixa laranja mesmo com o comando .click() e o movimento de arrastar "morreu no caminho", temos um conflito de camadas (layers) e foco.

Para resolver isso, vamos aplicar três ajustes de "força bruta":

Clique via JavaScript: Se o .click() normal do Selenium falha silenciosamente, o execute_script força o navegador a registrar o clique direto na raiz do elemento.

Double Check de Seleção: Vamos garantir que ele selecione a laranja, espere um pouco, e só então vá para as azuis.

Arraste com Offset de Segurança: Se ele "não chega até lá", vamos forçar o mouse a ultrapassar um pouco o centro do alvo para garantir que ele entre na zona de colisão.
